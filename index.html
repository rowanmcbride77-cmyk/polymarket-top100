const API = "https://data-api.polymarket.com"; const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", "Content-Type": "application/json; charset=utf-8", "Cache-Control": "no-store" }; function response(data, status = 200) { return new Response(JSON.stringify(data), { status, headers: cors }); } async function fetchJson(url) { const res = await fetch(url, { headers: { Accept: "application/json" } }); const text = await res.text(); if (!res.ok) { throw new Error(`Polymarket API ${res.status}: ${text.slice(0, 400)}`); } return JSON.parse(text); } function getArray(payload) { if (Array.isArray(payload)) return payload; if (Array.isArray(payload?.data)) return payload.data; return []; } function getWallet(row) { return ( row?.proxyWallet || row?.proxy_wallet || row?.user_id || row?.address || row?.user || "" ); } function getTimestamp(row) { const value = row?.timestamp ?? row?.createdAt ?? row?.created_at; if (value == null) return null; const n = Number(value); if (Number.isFinite(n)) return n; const parsed = Date.parse(value); return Number.isFinite(parsed) ? Math.floor(parsed / 1000) : null; } async function leaderboard(url) { const params = new URLSearchParams(url.searchParams); params.set("category", "OVERALL"); params.set("timePeriod", "WEEK"); params.set("orderBy", "PNL"); // The v1 leaderboard allows a maximum of 50 rows per request. params.set("limit", "50"); const firstOffset = Math.max(0, Number(params.get("offset") || 0)); params.set("offset", String(Math.min(firstOffset, 1000))); const first = await fetchJson( `${API}/v1/leaderboard?${params.toString()}` ); if (!Array.isArray(first)) { throw new Error("The official leaderboard did not return an array."); } // The existing page requests a leaderboard and displays up to 100. if (first.length < 50 || firstOffset >= 1000) { return first; } params.set("offset", String(firstOffset + 50)); const second = await fetchJson( `${API}/v1/leaderboard?${params.toString()}` ); return [ ...first, ...(Array.isArray(second) ? second : []) ]; } async function traderDetails(url) { const user = url.searchParams.get("user"); if (!user || !/^0x[a-fA-F0-9]{40}$/.test(user)) { throw new Error("A valid Polymarket wallet address is required."); } const now = Math.floor(Date.now() / 1000); const weekAgo = now - 7 * 24 * 60 * 60; const tradeParams = new URLSearchParams({ user, start: String(weekAgo), end: String(now), limit: "500", sort_by: "TIMESTAMP", sort_direction: "DESC" }); const positionParams = new URLSearchParams({ user, status: "OPEN", limit: "500", sort_by: "CURRENT_VALUE", sort_direction: "DESC" }); const [tradePayload, positionPayload] = await Promise.all([ fetchJson(`${API}/v2/trades?${tradeParams}`), fetchJson(`${API}/v2/positions?${positionParams}`) ]); const activity = getArray(tradePayload) .filter(row => { const timestamp = getTimestamp(row); return timestamp !== null && timestamp >= weekAgo && timestamp <= now; }) .map(row => ({ ...row, timestamp: getTimestamp(row), title: row.title || row.market || row.question || "Unknown market", outcome: row.outcome || "", side: row.side || "", size: row.size ?? null, price: row.price ?? null, amount: row.size != null && row.price != null ? Number(row.size) * Number(row.price) : null })); const positions = getArray(positionPayload); return { user, activity, positions, // These endpoints do not establish a complete, accurate weekly // resolved-market win/loss record. Never fabricate one from trades // or from positions that were merely closed. wins: null, losses: null, winRate: null, winLossAvailable: false }; } async function handle(request) { if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } if (request.method !== "GET") { return response({ error: "Method not allowed" }, 405); } const url = new URL(request.url); try { if (url.pathname === "/v1/leaderboard") { const data = await leaderboard(url); return response(data); } if (url.pathname === "/v2/trader") { const data = await traderDetails(url); return response(data); } if (url.pathname === "/positions") { const user = url.searchParams.get("user"); if (!user || !/^0x[a-fA-F0-9]{40}$/.test(user)) { return response({ error: "A valid user wallet is required." }, 400); } const params = new URLSearchParams({ user, status: "OPEN", limit: "500", sort_by: "CURRENT_VALUE", sort_direction: "DESC" }); const payload = await fetchJson(`${API}/v2/positions?${params}`); return response(getArray(payload)); } return response({ error: "Unknown endpoint" }, 404); } catch (error) { return response({ error: error instanceof Error ? error.message : String(error) }, 502); } } export default { fetch: handle };
