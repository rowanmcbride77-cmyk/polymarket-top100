/* ========================= SETTLED POSITIONS ========================= */ async function loadSettledStatus( trader, status ){ const start = periodStart(); const end = Math.floor( Date.now()/1000 ); let cursor = null; const rows = []; /* IMPORTANT: CLOSED = positions the trader exited. That is NOT the same thing as winning/losing a resolved market. REDEEMABLE = resolved winning positions. REDEEMABLE_LOST = resolved losing positions. We follow every cursor so large traders are not capped at the first page. */ while(true){ const params = new URLSearchParams({ user:trader.wallet, status:status, sort_by:'TIMESTAMP', sort_direction:'DESC', limit:'1000', start:String(start), end:String(end) }); if(cursor){ params.set( 'cursor', cursor ); } const result = await getJSON( DATA_API+ '/v2/positions?'+ params.toString() ); const page = Array.isArray( result?.data ) ? result.data : Array.isArray(result) ? result : []; rows.push( ...page ); const pagination = result?.pagination; const next = pagination?.next_cursor; const hasMore = pagination?.has_more; if( !hasMore || !next || next===cursor ){ break; } cursor = next; } return rows; } /* ========================= LOAD WEEKLY W/L ========================= */ async function loadClosedForTrader( trader ){ const [ winningRows, losingRows ] = await Promise.all([ loadSettledStatus( trader, 'REDEEMABLE' ), loadSettledStatus( trader, 'REDEEMABLE_LOST' ) ]); /* A trader can have multiple position rows for the same condition. W/L should represent the market decision, not count every token/position row. Therefore deduplicate by condition_id. */ const winsByCondition = new Map(); const lossesByCondition = new Map(); for( const position of winningRows ){ const condition = String( position?.condition_id || position?.conditionId || '' ); if(!condition) continue; if( !winsByCondition.has( condition ) ){ winsByCondition.set( condition, position ); } } for( const position of losingRows ){ const condition = String( position?.condition_id || position?.conditionId || '' ); if(!condition) continue; if( !lossesByCondition.has( condition ) ){ lossesByCondition.set( condition, position ); } } /* If the same condition appears in both sets, do not count it twice. Prefer the explicit winning settlement when available. */ const allConditions = new Set([ ...winsByCondition.keys(), ...lossesByCondition.keys() ]); let wins = 0; let losses = 0; let pnl = 0; let volume = 0; for( const condition of allConditions ){ const winningPosition = winsByCondition.get( condition ); const losingPosition = lossesByCondition.get( condition ); /* Explicit settlement status determines W/L. Do NOT use realized P/L to determine the record. */ if(winningPosition){ wins++; pnl += Number( winningPosition?.total_pnl ?? winningPosition?.realized_pnl ?? 0 ) || 0; volume += Number( winningPosition?.total_cost_usdc ?? winningPosition?.entry_cost_usdc ?? 0 ) || 0; }else if(losingPosition){ losses++; pnl += Number( losingPosition?.total_pnl ?? losingPosition?.realized_pnl ?? 0 ) || 0; volume += Number( losingPosition?.total_cost_usdc ?? losingPosition?.entry_cost_usdc ?? 0 ) || 0; } } trader.wins = wins; trader.losses = losses; trader.total = wins + losses; trader.winRate = trader.total > 0 ? wins / trader.total * 100 : 0; trader.periodPnl = trader.total > 0 ? pnl : trader.pnl; trader.periodVolume = volume; trader.recordsLoaded = true; }
